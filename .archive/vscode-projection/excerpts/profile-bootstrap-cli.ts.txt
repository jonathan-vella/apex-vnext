function profileClientId(flags: Flags): "github-copilot-vscode" {
  const value = flags.client ?? "github-copilot-vscode";
  if (value !== "github-copilot-vscode") {
    throw new ApexError(
      "APEX_USAGE",
      "Profile bootstrap is supported only for github-copilot-vscode",
      EXIT_CODES.usage,
    );
  }
  return value;
}
